4. CREATE VIEW "daily_kwh" AS
     SELECT m.site_id, date_trunc('day', r.ts)::date AS day,
            sum(r.kwh) AS total_kwh, max(r.kw_peak) AS peak_kw
       FROM readings r JOIN meters m ON m.meter_id = r.meter_id
      GROUP BY 1, 2;
